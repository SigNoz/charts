{{- if .Values.signoz.ingress.enabled }}

Open SigNoz at:
{{- range $host := .Values.signoz.ingress.hosts }}
{{- $scheme := "http" }}
{{- range $.Values.signoz.ingress.tls }}
{{- if has $host.host .hosts }}{{ $scheme = "https" }}{{ end }}
{{- end }}
{{- range .paths }}
  {{ $scheme }}://{{ $host.host }}{{ .path }}
{{- end }}
{{- end }}
{{- else if contains "NodePort" .Values.signoz.service.type }}

Get the SigNoz URL:

  export NODE_PORT=$(kubectl get --namespace {{ include "signoz.namespace" . }} services {{ include "signoz.fullname" . }} -o jsonpath='{.spec.ports[?(@.name=="http")].nodePort}')
  export NODE_IP=$(kubectl get nodes -o jsonpath='{.items[0].status.addresses[?(@.type=="InternalIP")].address}')
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.signoz.service.type }}

Once the load balancer has an address, get the SigNoz URL:

  export SERVICE_HOST=$(kubectl get --namespace {{ include "signoz.namespace" . }} services {{ include "signoz.fullname" . }} -o jsonpath='{.status.loadBalancer.ingress[0].ip}{.status.loadBalancer.ingress[0].hostname}')
  echo http://$SERVICE_HOST:{{ .Values.signoz.service.port }}
{{- else if contains "ClusterIP" .Values.signoz.service.type }}

Forward a local port to SigNoz, then open http://127.0.0.1:{{ .Values.signoz.service.port }}:

  export POD_NAME=$(kubectl get pods --namespace {{ include "signoz.namespace" . }} -l "{{ include "signoz.selectorLabels" . | replace ": " "=" | replace "\n" "," }}" -o jsonpath="{.items[0].metadata.name}")
  kubectl --namespace {{ include "signoz.namespace" . }} port-forward $POD_NAME {{ .Values.signoz.service.port }}:8080
{{- end }}
